# NOTE:
# File nay chi la mau tham khao/chua phai file Dockerfile chay truc tiep.
# Docker Compose hien dung file that tai: server/Dockerfile
#
# ==============================
# 1. Chọn base image Node.js
# Node 20+ là bắt buộc vì dependencies hiện tại yêu cầu Node >=20.19.0
# alpine = nhẹ hơn (khuyên dùng)
# ==============================
FROM node:20-alpine

# ==============================
# 2. Tạo thư mục làm việc trong container
# ==============================
WORKDIR /app

# ==============================
# 3. Copy package.json trước
# để tận dụng cache khi build
# ==============================
COPY package*.json ./

# ==============================
# 4. Cài dependencies
# ==============================
RUN npm ci

# ==============================
# 5. Copy toàn bộ source code
# ==============================
COPY . .

# ==============================
# 6. Mở port (phải trùng server.js)
# ==============================
EXPOSE 5000

# ==============================
# 7. Chạy app
# ==============================
CMD ["npm", "start"]
